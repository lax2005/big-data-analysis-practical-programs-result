CREATE TABLE matrix_a ( 
r INT, 
c INT, 
val INT 
) 
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY ',';


CREATE TABLE matrix_b ( 
r INT, 
c INT, 
val INT 
) 
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY ','; 


CREATE TABLE matrix_c ( 
r INT, 
c INT, 
val INT    
) 
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY ',';

INSERT OVERWRITE TABLE matrix_c 
         SELECT 
             a.r, 
             b.c, 
         SUM(a.val * b.val) AS val 
         FROM matrix_a a 
         JOIN matrix_b b 
         ON a.c = b.r 
         GROUP BY a.r, b.c;
